  3          SELECT book_id, title, stock
  4          FROM book
  5          WHERE category = 'Database'
  6          FOR UPDATE OF stock;
  7  
  8      book_rec book_cursor%ROWTYPE;
  9      v_old_stock NUMBER;
 10      v_new_stock NUMBER;
 11  BEGIN
 12      OPEN book_cursor;
 13  
 14      LOOP
 15          FETCH book_cursor INTO book_rec;
 16  
 17          EXIT WHEN book_cursor%NOTFOUND;

 19          v_old_stock := book_rec.stock;
